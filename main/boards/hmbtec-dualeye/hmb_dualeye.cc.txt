// ============================================================================
// SPI Low Level
// ============================================================================
static void txBytes(spi_device_handle_t d,const void* data,size_t len){
    spi_transaction_t t={};
    t.length=len*8;
    t.tx_buffer=data;
    ESP_ERROR_CHECK(spi_device_polling_transmit(d,&t));
}
static void cmd(spi_device_handle_t d,uint8_t c){
    gpio_set_level(HMB_EYE_DC,0);
    txBytes(d,&c,1);
}
static void data(spi_device_handle_t d,const void* p,size_t n){
    gpio_set_level(HMB_EYE_DC,1);
    txBytes(d,p,n);
}
static void cd(spi_device_handle_t d,uint8_t c,const uint8_t* p,size_t n){
    cmd(d,c);
    if(n) data(d,p,n);
}
static void write8(spi_device_handle_t d,uint8_t command,uint8_t value){
    cd(d,command,&value,1);
}

// ============================================================================
// Primitive Grafikfunktionen
// ============================================================================
static void hline(int x,int y,int w,uint16_t c){
    if(y<0 || y>=H || w<=0) return;
    if(x<0){
        w+=x;
        x=0;
    }
    if(x+w>W) w=W-x;
    if(w<=0) return;
    for(int i=0;i<w;i++) fb[y*W+x+i]=c;
}
static void circle(int cx,int cy,int r,uint16_t c){
    int r2=r*r;
    for(int y=-r;y<=r;y++){
        int yy=r2-y*y;
        if(yy<0) continue;
        int x=(int)sqrtf((float)yy);
        hline(cx-x,cy+y,x*2+1,c);
    }
}

// ============================================================================
// Expression Masken
// ============================================================================
static void maskTopFlat(int depth){
    for(int y=0;y<depth;y++) hline(0,y,W,BLACK);
}
static void maskBottomFlat(int depth){
    for(int y=0;y<depth;y++) hline(0,H-1-y,W,BLACK);
}
static void maskTopSlope(int baseDepth,int slope,bool mirror){
    for(int x=0;x<W;x++){
        float nx=(float)x/(float)(W-1);
        if(mirror) nx=1.0f-nx;
        int depth=baseDepth+(int)((nx-0.5f)*(float)slope);
        depth=std::max(0,std::min(H,depth));
        for(int y=0;y<depth;y++) fb[y*W+x]=BLACK;
    }
}

// ============================================================================
// Auge rendern
// ============================================================================
static void render(bool leftEye,HmbEyeExpression expression){
    std::fill_n(fb,N,BLACK);

    float ex=px;
    float ey=py;
    int pupil=PUPIL_RADIUS;

    switch(expression){
        case HmbEyeExpression::Thinking:
            ex+=(leftEye ? -0.15f : 0.15f)*LOOK_X_MAX;
            ey-=0.45f*LOOK_Y_MAX;
            break;
        case HmbEyeExpression::Curious:
            ex+=(leftEye ? -0.12f : 0.12f)*LOOK_X_MAX;
            ey-=0.18f*LOOK_Y_MAX;
            break;
        case HmbEyeExpression::Surprised:
            ex=0.0f;
            ey=0.0f;
            pupil=(int)(PUPIL_RADIUS*1.18f);
            break;
        case HmbEyeExpression::Sad:
            ey+=0.28f*LOOK_Y_MAX;
            break;
        case HmbEyeExpression::Confused:
            ex+=(leftEye ? -0.20f : 0.20f)*LOOK_X_MAX;
            break;
        case HmbEyeExpression::Sleepy:
            ey+=0.15f*LOOK_Y_MAX;
            break;
        default:
            break;
    }

    ex=std::max(-LOOK_X_MAX,std::min(LOOK_X_MAX,ex));
    ey=std::max(-LOOK_Y_MAX,std::min(LOOK_Y_MAX,ey));

    circle(EYE_CX,EYE_CY,EYEBALL_RADIUS,WHITE);

    int ix=EYE_CX+(int)ex;
    int iy=EYE_CY+(int)ey;

    circle(ix,iy,IRIS_RADIUS_1,IRIS1);
    circle(ix,iy,IRIS_RADIUS_2,IRIS2);
    circle(ix,iy,IRIS_RADIUS_3,IRIS3);
    circle(ix,iy,pupil,BLACK);
    circle(ix+HIGHLIGHT1_X,iy+HIGHLIGHT1_Y,HIGHLIGHT1_R,WHITE);
    circle(ix+HIGHLIGHT2_X,iy+HIGHLIGHT2_Y,HIGHLIGHT2_R,WHITE);

    // Semantische Lidstellung
    switch(expression){
        case HmbEyeExpression::Happy:
            maskBottomFlat((int)(H*0.13f));
            maskTopFlat((int)(H*0.05f));
            break;
        case HmbEyeExpression::Relaxed:
            maskTopFlat((int)(H*0.11f));
            maskBottomFlat((int)(H*0.04f));
            break;
        case HmbEyeExpression::Curious:
            maskTopFlat(leftEye ? (int)(H*0.03f) : (int)(H*0.08f));
            break;
        case HmbEyeExpression::Thinking:
            maskTopSlope((int)(H*0.07f),(int)(H*0.08f),leftEye);
            break;
        case HmbEyeExpression::Sad:
            maskTopSlope((int)(H*0.10f),(int)(H*0.15f),!leftEye);
            break;
        case HmbEyeExpression::Confused:
            if(leftEye) maskTopSlope((int)(H*0.05f),(int)(H*0.12f),false);
            else maskTopFlat((int)(H*0.12f));
            break;
        case HmbEyeExpression::Sleepy:
            maskTopFlat((int)(H*0.27f));
            maskBottomFlat((int)(H*0.08f));
            break;
        default:
            break;
    }

    // Natürliches Blinzeln liegt über jeder Expression
    int lid=(int)((float)BLINK_DEPTH*blink);
    for(int y=0;y<lid;y++){
        float f=1.0f-(float)y/(float)BLINK_DEPTH;
        int in=(int)((float)W*0.0625f*f);
        hline(in,y,W-in*2,BLACK);
        hline(in,H-1-y,W-in*2,BLACK);
    }
}

// ============================================================================
// Frame an Display übertragen
// ============================================================================
static void push(spi_device_handle_t d){
    uint16_t xEnd=W-1;
    uint16_t yEnd=H-1;
    const uint8_t col[]={0x00,0x00,(uint8_t)(xEnd>>8),(uint8_t)(xEnd&0xFF)};
    const uint8_t row[]={0x00,0x00,(uint8_t)(yEnd>>8),(uint8_t)(yEnd&0xFF)};
    cd(d,0x2A,col,sizeof(col));
    cd(d,0x2B,row,sizeof(row));
    cmd(d,0x2C);
    gpio_set_level(HMB_EYE_DC,1);
    static uint8_t line[W*2];
    for(int y=0;y<H;y++){
        for(int x=0;x<W;x++){
            uint16_t c=fb[y*W+x];
            line[x*2]=(uint8_t)(c>>8);
            line[x*2+1]=(uint8_t)(c&0xFF);
        }
        txBytes(d,line,sizeof(line));
    }
}

static int randomSigned(int minimum,int maximum){
    uint32_t range=(uint32_t)(maximum-minimum+1);
    return (int)(esp_random()%range)+minimum;
}
static const char* expressionName(HmbEyeExpression e){
    switch(e){
        case HmbEyeExpression::Happy: return "happy";
        case HmbEyeExpression::Relaxed: return "relaxed";
        case HmbEyeExpression::Curious: return "curious";
        case HmbEyeExpression::Thinking: return "thinking";
        case HmbEyeExpression::Surprised: return "surprised";
        case HmbEyeExpression::Sad: return "sad";
        case HmbEyeExpression::Confused: return "confused";
        case HmbEyeExpression::Sleepy: return "sleepy";
        default: return "neutral";
    }
}

static const char* lookName(HmbEyeLook look){
    switch(look){
        case HmbEyeLook::Center: return "center";
        case HmbEyeLook::Left: return "left";
        case HmbEyeLook::Right: return "right";
        case HmbEyeLook::Up: return "up";
        case HmbEyeLook::Down: return "down";
        case HmbEyeLook::UpLeft: return "up_left";
        case HmbEyeLook::UpRight: return "up_right";
        case HmbEyeLook::DownLeft: return "down_left";
        case HmbEyeLook::DownRight: return "down_right";
        default: return "auto";
    }
}
static void applyLookTarget(HmbEyeLook look){
    switch(look){
        case HmbEyeLook::Center: tx=0.0f; ty=0.0f; break;
        case HmbEyeLook::Left: tx=-LOOK_X_MAX; ty=0.0f; break;
        case HmbEyeLook::Right: tx=LOOK_X_MAX; ty=0.0f; break;
        case HmbEyeLook::Up: tx=0.0f; ty=-LOOK_Y_MAX; break;
        case HmbEyeLook::Down: tx=0.0f; ty=LOOK_Y_MAX; break;
        case HmbEyeLook::UpLeft: tx=-LOOK_X_MAX*0.82f; ty=-LOOK_Y_MAX*0.82f; break;
        case HmbEyeLook::UpRight: tx=LOOK_X_MAX*0.82f; ty=-LOOK_Y_MAX*0.82f; break;
        case HmbEyeLook::DownLeft: tx=-LOOK_X_MAX*0.82f; ty=LOOK_Y_MAX*0.82f; break;
        case HmbEyeLook::DownRight: tx=LOOK_X_MAX*0.82f; ty=LOOK_Y_MAX*0.82f; break;
        default: break;
    }
}

// ============================================================================
// Eye Animation Task
// ============================================================================
static void task(void*){
    nextLook=esp_timer_get_time()/1000+800;
    nextBlink=esp_timer_get_time()/1000+3000;

    while(true){
        int64_t now=esp_timer_get_time()/1000;
        DeviceState s=eyeState;
        HmbEyeExpression expression=eyeExpression;
        HmbEyeLook look=eyeLook;

        if(look!=HmbEyeLook::Auto && lookUntil>0 && now>=lookUntil){
            eyeLook=HmbEyeLook::Auto;
            lookUntil=0;
            look=HmbEyeLook::Auto;
            nextLook=now;
            ESP_LOGI(TAG,"Look timeout -> auto");
        }

        if(expression!=HmbEyeExpression::Neutral && expressionUntil>0 && now>=expressionUntil){
            eyeExpression=HmbEyeExpression::Neutral;
            expressionUntil=0;
            expression=HmbEyeExpression::Neutral;
            ESP_LOGI(TAG,"Expression timeout -> neutral");
        }

        // Explizite Blickrichtung hat Vorrang vor Expression und DeviceState.
        if(look!=HmbEyeLook::Auto){
            applyLookTarget(look);
        }
        // Expression beeinflusst Blickverhalten, DeviceState bleibt Fallback.
        else if(expression==HmbEyeExpression::Surprised){
            tx=0.0f;
            ty=0.0f;
        }
        else if(expression==HmbEyeExpression::Thinking){
            if(now>=nextLook){
                tx=(float)randomSigned(-(int)(LOOK_X_MAX*0.35f),(int)(LOOK_X_MAX*0.35f));
                ty=-(LOOK_Y_MAX*0.55f);
                nextLook=now+1200;
            }
        }
        else if(expression==HmbEyeExpression::Curious){
            if(now>=nextLook){
                tx=(float)randomSigned(-(int)(LOOK_X_MAX*0.55f),(int)(LOOK_X_MAX*0.55f));
                ty=(float)randomSigned(-(int)(LOOK_Y_MAX*0.30f),(int)(LOOK_Y_MAX*0.15f));
                nextLook=now+900;
            }
        }
        else if(expression==HmbEyeExpression::Sad){
            if(now>=nextLook){
                tx=(float)randomSigned(-(int)(LOOK_X_MAX*0.25f),(int)(LOOK_X_MAX*0.25f));
                ty=LOOK_Y_MAX*0.35f;
                nextLook=now+1400;
            }
        }
        else if(expression==HmbEyeExpression::Confused){
            if(now>=nextLook){
                tx=(float)randomSigned(-(int)(LOOK_X_MAX*0.45f),(int)(LOOK_X_MAX*0.45f));
                ty=(float)randomSigned(-(int)(LOOK_Y_MAX*0.20f),(int)(LOOK_Y_MAX*0.20f));
                nextLook=now+650;
            }
        }
        else if(expression==HmbEyeExpression::Sleepy){
            if(now>=nextLook){
                tx=(float)randomSigned(-(int)(LOOK_X_MAX*0.18f),(int)(LOOK_X_MAX*0.18f));
                ty=LOOK_Y_MAX*0.20f;
                nextLook=now+1800;
            }
        }
        else if(expression==HmbEyeExpression::Relaxed){
            if(now>=nextLook){
                tx=(float)randomSigned(-(int)(LOOK_X_MAX*0.22f),(int)(LOOK_X_MAX*0.22f));
                ty=(float)randomSigned(-(int)(LOOK_Y_MAX*0.12f),(int)(LOOK_Y_MAX*0.18f));
                nextLook=now+1300;
            }
        }
        else if(expression==HmbEyeExpression::Happy){
            if(now>=nextLook){
                tx=(float)randomSigned(-(int)(LOOK_X_MAX*0.45f),(int)(LOOK_X_MAX*0.45f));
                ty=(float)randomSigned(-(int)(LOOK_Y_MAX*0.25f),(int)(LOOK_Y_MAX*0.25f));
                nextLook=now+750;
            }
        }
        else if(s==kDeviceStateListening){
            tx=0.0f;
            ty=0.0f;
        }
        else if(s==kDeviceStateConnecting || s==kDeviceStateWifiConfiguring){
            if(now>=nextLook){
                int lookX=(int)LOOK_X_MAX;
                int lookY=(int)(LOOK_Y_MAX*0.55f);
                tx=(float)randomSigned(-lookX,lookX);
                ty=(float)randomSigned(-lookY,lookY);
                nextLook=now+500;
            }
        }
        else if(s==kDeviceStateSpeaking){
            if(now>=nextLook){
                int lookX=(int)(LOOK_X_MAX*0.65f);
                int lookY=(int)(LOOK_Y_MAX*0.55f);
                tx=(float)randomSigned(-lookX,lookX);
                ty=(float)randomSigned(-lookY,lookY);
                nextLook=now+700;
            }
        }
        else if(now>=nextLook){
            int lookX=(int)LOOK_X_MAX;
            int lookY=(int)LOOK_Y_MAX;
            tx=(float)randomSigned(-lookX,lookX);
            ty=(float)randomSigned(-lookY,lookY);
            nextLook=now+800+(int64_t)(esp_random()%1800);
        }

        px+=(tx-px)*0.18f;
        py+=(ty-py)*0.18f;
        px=std::max(-LOOK_X_MAX,std::min(LOOK_X_MAX,px));
        py=std::max(-LOOK_Y_MAX,std::min(LOOK_Y_MAX,py));

        if(now>=nextBlink && !closing && blink<=0.0f){
            closing=true;
            int blinkBase=(expression==HmbEyeExpression::Sleepy) ? 1800 : 2500;
            int blinkRange=(expression==HmbEyeExpression::Sleepy) ? 1800 : 3500;
            nextBlink=now+blinkBase+(int64_t)(esp_random()%blinkRange);
        }
        if(closing){
            blink+=0.32f;
            if(blink>=1.0f){
                blink=1.0f;
                closing=false;
            }
        }
        else if(blink>0.0f){
            blink-=0.24f;
            if(blink<0.0f) blink=0.0f;
        }

        // Links und rechts getrennt rendern -> echte asymmetrische Expressions
        render(true,expression);
        push(devL);
        render(false,expression);
        push(devR);

        vTaskDelay(pdMS_TO_TICKS(45));
    }
}

void HmbDualEye::Init(){
    ESP_LOGI(TAG,"Initializing HMBTEC DualEye");
#if HMB_EYE_DISPLAY == HMB_EYE_GC9D01_160
    ESP_LOGI(TAG,"Display: GC9D01 160x160");
#elif HMB_EYE_DISPLAY == HMB_EYE_GC9A01_240
    ESP_LOGI(TAG,"Display: GC9A01 240x240");
#endif
    ESP_LOGI(TAG,"Framebuffer: %dx%d = %u bytes",W,H,(unsigned)(N*sizeof(uint16_t)));

    fb=(uint16_t*)heap_caps_malloc(N*sizeof(uint16_t),MALLOC_CAP_SPIRAM|MALLOC_CAP_8BIT);
    if(!fb){
        ESP_LOGE(TAG,"Could not allocate DualEye framebuffer: %u bytes",(unsigned)(N*sizeof(uint16_t)));
        return;
    }
    std::memset(fb,0,N*sizeof(uint16_t));

    gpio_config_t g={};
    g.pin_bit_mask=(1ULL<<HMB_EYE_DC)|(1ULL<<HMB_EYE_RST_LEFT)|(1ULL<<HMB_EYE_RST_RIGHT);
    g.mode=GPIO_MODE_OUTPUT;
    ESP_ERROR_CHECK(gpio_config(&g));

    gpio_set_level(HMB_EYE_RST_LEFT,0);
    gpio_set_level(HMB_EYE_RST_RIGHT,0);
    vTaskDelay(pdMS_TO_TICKS(30));
    gpio_set_level(HMB_EYE_RST_LEFT,1);
    gpio_set_level(HMB_EYE_RST_RIGHT,1);
    vTaskDelay(pdMS_TO_TICKS(150));

    spi_bus_config_t bus={};
    bus.mosi_io_num=HMB_EYE_MOSI;
    bus.miso_io_num=-1;
    bus.sclk_io_num=HMB_EYE_SCLK;
    bus.quadwp_io_num=-1;
    bus.quadhd_io_num=-1;
    bus.max_transfer_sz=W*2;
    ESP_ERROR_CHECK(spi_bus_initialize(SPI2_HOST,&bus,SPI_DMA_CH_AUTO));

    spi_device_interface_config_t di={};
    di.clock_speed_hz=20000000;
    di.mode=0;
    di.spics_io_num=HMB_EYE_CS_LEFT;
    di.queue_size=1;
    ESP_ERROR_CHECK(spi_bus_add_device(SPI2_HOST,&di,&devL));

    di.spics_io_num=HMB_EYE_CS_RIGHT;
    ESP_ERROR_CHECK(spi_bus_add_device(SPI2_HOST,&di,&devR));

    initPanel(devL);
    initPanel(devR);

    BaseType_t result=xTaskCreate(task,"hmb_eye",6144,nullptr,2,nullptr);
    if(result!=pdPASS){
        ESP_LOGE(TAG,"Could not create DualEye task");
        heap_caps_free(fb);
        fb=nullptr;
        return;
    }
    ESP_LOGI(TAG,"DualEye ready");
}

bool HmbDualEye::SetExpression(const std::string& expression,int duration_ms){
    HmbEyeExpression e=HmbEyeExpression::Neutral;

    if(expression=="neutral") e=HmbEyeExpression::Neutral;
    else if(expression=="happy") e=HmbEyeExpression::Happy;
    else if(expression=="relaxed") e=HmbEyeExpression::Relaxed;
    else if(expression=="curious") e=HmbEyeExpression::Curious;
    else if(expression=="thinking") e=HmbEyeExpression::Thinking;
    else if(expression=="surprised") e=HmbEyeExpression::Surprised;
    else if(expression=="sad") e=HmbEyeExpression::Sad;
    else if(expression=="confused") e=HmbEyeExpression::Confused;
    else if(expression=="sleepy") e=HmbEyeExpression::Sleepy;
    else{
        ESP_LOGW(TAG,"Unknown expression: %s",expression.c_str());
        return false;
    }

    duration_ms=std::max(500,std::min(15000,duration_ms));
    eyeExpression=e;

    if(e==HmbEyeExpression::Neutral){
        expressionUntil=0;
    }
    else{
        expressionUntil=esp_timer_get_time()/1000+duration_ms;
    }

    ESP_LOGI(TAG,"Expression: %s, duration=%d ms",expressionName(e),duration_ms);
    return true;
}
void HmbDualEye::ClearExpression(){
    eyeExpression=HmbEyeExpression::Neutral;
    expressionUntil=0;
    ESP_LOGI(TAG,"Expression cleared -> neutral");
}
bool HmbDualEye::SetLook(const std::string& direction,int duration_ms){
    HmbEyeLook look=HmbEyeLook::Auto;
    if(direction=="auto") look=HmbEyeLook::Auto;
    else if(direction=="center") look=HmbEyeLook::Center;
    else if(direction=="left") look=HmbEyeLook::Left;
    else if(direction=="right") look=HmbEyeLook::Right;
    else if(direction=="up") look=HmbEyeLook::Up;
    else if(direction=="down") look=HmbEyeLook::Down;
    else if(direction=="up_left") look=HmbEyeLook::UpLeft;
    else if(direction=="up_right") look=HmbEyeLook::UpRight;
    else if(direction=="down_left") look=HmbEyeLook::DownLeft;
    else if(direction=="down_right") look=HmbEyeLook::DownRight;
    else{
        ESP_LOGW(TAG,"Unknown look direction: %s",direction.c_str());
        return false;
    }
    duration_ms=std::max(500,std::min(15000,duration_ms));
    eyeLook=look;
    if(look==HmbEyeLook::Auto){
        lookUntil=0;
    }
    else{
        lookUntil=esp_timer_get_time()/1000+duration_ms;
        applyLookTarget(look);
    }
    ESP_LOGI(TAG,"Look: %s, duration=%d ms",lookName(look),duration_ms);
    return true;
}
void HmbDualEye::ClearLook(){
    eyeLook=HmbEyeLook::Auto;
    lookUntil=0;
    ESP_LOGI(TAG,"Look cleared -> auto");
}
